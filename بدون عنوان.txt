   عرض الأذكار
========================= */

function renderSection(section){

const list=data[section];

const box=document.getElementById(section+"List");

box.innerHTML="";

list.forEach(function(item,index){

const key=section+"_"+index;

let value=Number(localStorage.getItem(key)||0);

if(value>item[1]) value=item[1];

const div=document.createElement("div");

div.className="zikr";

let readButton="";

if(surahs[item[0]]){

readButton=
'<button class="read" data-read="'+index+'">📖 اقرأ</button>'+
'<div class="surahText" id="surah_'+section+'_'+index+'">'+
surahs[item[0]]+
'</div>';

}

div.innerHTML=
'<div class="num">الذكر '+(index+1)+' من '+list.length+'</div>'+
'<div class="text">'+item[0]+'</div>'+
readButton+
'<div class="buttons">'+
'<div class="count" id="count_'+key+'">'+value+' / '+item[1]+'</div>'+
'<button class="plus" data-plus="'+index+'">اضغط للذكر</button>'+
'<button class="repeat" data-repeat="'+index+'" title="إعادة هذا الذكر">🔄</button>'+
'</div>';

box.appendChild(div);

});


/* زر الزيادة */

box.querySelectorAll("[data-plus]").forEach(function(button){

button.addEventListener("click",function(){

const index=Number(button.dataset.plus);

const key=section+"_"+index;

let value=Number(localStorage.getItem(key)||0);

if(value<list[index][1]){

value++;

localStorage.setItem(key,value);

}

document.getElementById("count_"+key).textContent=
value+" / "+list[index][1];


/* عند اكتمال كل الأذكار */

let finished=list.every(function(item,i){

return Number(localStorage.getItem(section+"_"+i)||0)>=item[1];

});

if(finished){

setTimeout(function(){

alert("🎉 انتهيت من الأذكار! ستبدأ من الأول.");

resetSection(section);

},200);

}

});

});


/* زر إعادة الذكر الواحد */

box.querySelectorAll("[data-repeat]").forEach(function(button){

button.addEventListener("click",function(){

const index=Number(button.dataset.repeat);

const key=section+"_"+index;

localStorage.removeItem(key);

document.getElementById("count_"+key).textContent=
"0 / "+list[index][1];

});

});


/* زر قراءة السورة */

box.querySelectorAll("[data-read]").forEach(function(button){

button.addEventListener("click",function(){

const index=Number(button.dataset.read);

const text=document.getElementById(
"surah_"+section+"_"+index
);

if(text.style.display==="block"){

text.style.display="none";

button.textContent="📖 اقرأ";

}else{

text.style.display="block";

button.textContent="🔽 إخفاء";

}

});

});

}

function resetSection(section){

data[section].forEach(function(_,index){

async function getSurahs(){

const select=document.getElementById("surahSelect");

try{

const response=
await fetch("https://api.alquran.cloud/v1/surah");

const json=await response.json();

select.innerHTML=
'<option value="">اختر السورة</option>';

json.data.forEach(function(surah){

const option=document.createElement("option");

option.value=surah.number;

option.textContent=
surah.number+" - "+surah.name;

select.appendChild(option);

});

}catch(error){

select.innerHTML=
'<option>تعذر تحميل السور — تحقق من الإنترنت</option>';

}

}


document.getElementById("surahSelect")
.addEventListener("change",async function(){

const number=this.value;

if(!number)return;

const content=document.getElementById("quranContent");

content.innerHTML="⏳ جاري تحميل السورة...";

try{

const response=
await fetch(
"https://api.alquran.cloud/v1/surah/"
+number+
"/quran-uthmani"
);

const json=await response.json();

const surah=json.data;

let html=
'<div class="surahTitle">'+
surah.name+
'</div>';

surah.ayahs.forEach(function(ayah){

html+=
'<div class="ayah">'+
ayah.text+
' ﴿'+ayah.numberInSurah+'﴾'+
'</div>';

});

content.innerHTML=html;

}catch(error){

content.innerHTML=
"❌ لم يتم تحميل السورة. تأكد من اتصال الإنترنت.";

}

});
